The best I can do with the code I have is get Ambient settings to kind-of work from the material, but it doesn't appear to pick up Diffuse or Specular--and ultimately, I'd like to be able to add my own shader. My verts have positions, normals, and tangents--which all appear to be correct.

Code: Select all
void RibbonGlob::Preprocess(void)
{
   SetEffectListIndex(C4::kEffectListLight);
   C4::Effect::Preprocess();
   Wake();
   
   SetTransformable(this);
      
   C4::MaterialObject *material = C4::MaterialObject::Get("player/RibbonGlob");
   if (!material) {
      C4::TheEngine->Report("There is no material for ribbon glob!");
   }
   else {
      SetMaterialObject(material);
      
      C4::RenderSegment *segment = GetFirstRenderSegment();
      segment->SetMaterialObjectPointer(&materialObject);
      segment->SetMaterialAttributeList(materialObject->GetAttributeList());
      segment->SetMaterialState(C4::kMaterialOcclusionInhibit | C4::kMaterialAlphaTest);
      SetLightBlendState(C4::BlendState(C4::kBlendSourceAlpha, C4::kBlendOne));
   }
}

by Eric Lengyel » 21 Nov 2013, 16:41

Your code looks like it should be fine. The only thing I see at the moment is that your light-pass blending mode multiplies the source color by the source alpha. Are you sure that your shader is outputting a nonzero alpha value? Perhaps try SetBlendMode(kBlendAccumulate) to see if anything shows up.

   SetTransformable(this);
   SetTransparentPosition(&GetWorldPosition());
   
   if (GetManipulator()){
      return;
   }
   
   C4::MaterialObject *material = C4::MaterialObject::Get("player/RibbonGlob");
   if (!material) {
      C4::TheEngine->Report("There is no material for ribbon glob!");

      C4::RenderSegment *segment = GetFirstRenderSegment();
      segment->SetMaterialObjectPointer(&materialObject);

      segment->SetMaterialState(C4::kMaterialOcclusionInhibit); // | C4::kMaterialAlphaTest
      
      SetShaderFlags(GetShaderFlags() & ~C4::kShaderAmbientEffect); //from BlobParticlesystem
