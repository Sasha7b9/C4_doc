   }

   void Preprocess(void)

      SetCollisionKind(GetCollisionKind() | kCollisionShot);

      SetRigidBodyFlags(GetRigidBodyFlags() | kRigidBodyKeepAwake | kRigidBodyFixedOrientation | kRigidBodyDisabledContact);
      SetGravityMultiplier(0.0F);

      SetLinearVelocity(speed);

   void Move(void)

      return (RigidBodyController::ValidRigidBodyCollision(body));
   }

   void SendInitialStateMessages(Player *player) const override

            ExplosionParticleSystem * explosion = new ExplosionParticleSystem();
            Node * target = GetTargetNode();
            explosion->SetNodePosition(target->GetOwningZone()->GetInverseWorldTransform() * msg->GetPosition());
            target->GetOwningZone()->AddNewSubnode(explosion);

            Destroy();
         }
         break;
      default:
         {
            ProjectileController::ReceiveMessage(message);
            break;
         }
      }

   ControllerMessage *ConstructMessage(ControllerMessageType type) const override
   {
      switch(type)
      {

      default:
         return ProjectileController::ConstructMessage(type);

   RigidBodyStatus HandleContact()

Re: Weird problems with RigidBody collisions

by Eric Lengyel » 16 Jun 2013, 18:37

Shelim wrote:What is correct behavior for rigid body controller when it's already overlapping with another geometry during preprocess? Is it undefined and could result in my situation by any chance?


In this case, the constraint solver would attempt to separate the rigid body from the geometry so that they are no longer intersecting. This could result in the rigid body shooting off in a random direction at high speed, but once the rigid body is separated from the geometry, it should otherwise behave normally.

Re: Weird problems with RigidBody collisions

by shelim » 23 Jun 2013, 16:16

I've got it! And it indeed was so stupid that I'm feeling ashamed...

It was:
Code: Select all
// Called before preprocess
projectile->SetNodeTransform(Transform4D(Matrix3D(front, up, left), zone->GetInverseWorldTransform() * GetPosition()));
and should be:
Code: Select all
 // Called before preprocess
projectile->SetNodeTransform(Transform4D(Matrix3D(front, left, up), zone->GetInverseWorldTransform() * GetPosition()));
(Vectors building up Matrix3D in wrong order)

I would notice this much earlier if my missile geometry was not symetric around it's axis...
