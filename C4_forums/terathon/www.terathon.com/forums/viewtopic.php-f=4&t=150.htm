Re: Collada 1.3 exporter question

by Eric Lengyel » 23 May 2006, 05:24

Huidafa wrote:Should I be using the "triangles" element type instead?


The Collada Importer treats the <polygons> and <triangles> elements equally -- it doesn't care which one you use.

Huidafa wrote:Also, is C4 paying attention to the normals or should I leave those out?


The normals are currently ignored, but the 1.4 importer will have the option of using the normals in the Collada file unmodified.

Huidafa wrote:Finally, can you explain this interleave compression scheme used in the sample Collada file for a cube


In that example, since there are only 8 unique vertex positions, they write out a samller array. You're not required to do that, and the importer would have to expand them anyway. I'm not sure why they've written out 24 normals, since there are only 6 unique ones. It doesn't really matter, though. You can do anything that will ultimately associate the correct three vertices, three normals, and three texcoords with a triangle (or four for a quad). C4 will handle the rest and automatically collapse any duplicates (matching in all attributes) into the same vertex.

Re: Collada 1.3 exporter question (follow up)

by Eric Lengyel » 23 May 2006, 05:56

Huidafa wrote:Thanks. So I can just write out the polygon stream using triangles, without bothering to duplicate the last vertex of each triangle to form a quad?


Yes. Creating a degenerate vertex to fill out a quad would be bad. In a <p> element, just write out the number of vertices that make up the polygon, whether it be 3, 4, or more. (The engine can handle any number of vertices in a polygon, but it's your responsibility to make sure they're all coplanar.) You can also mix different sized polygons inside a single <polygons> element.

That did the trick. Thanks for the tip, Eric.

I just successfully exported a cube into C4 via Collada 1.3 from Carrara 5.1. (Although it should work with earlier versions of Carrara.) Here is the main section of the export function:

Code: Select all
   TMCCountedPtr<I3DShPrimitive>   primitive;
   TMCCountedPtr<I3DShObject>      object;
   anInstance->Get3DObject(&object);
   object->QueryInterface(IID_I3DShPrimitive, (void**) &primitive);
   if (primitive)
   {
      TMCCountedPtr<I3DShTreeElement>   tree;
      anInstance->QueryInterface(IID_I3DShTreeElement, (void**) &tree);
      // Calc Name
      TMCString1023 name;
      TMCString31 indexString;
      indexString.FromInt32((int32)count);
      tree->GetName(name);
      if (name.Length() == 0)
      {
         name = "prim\0";
      }
      name += indexString;

      int8 p[1024];
      name.ToCPtr(p);

      int i;
      for (i=0; i<1000 && p[i]!=0; i++)
      {   // Make the name friendly
         if (((p[i]<'a') || (p[i]>'z')) &&
            ((p[i]<'A') || (p[i]>'Z')) &&
            ((p[i]<'0') || (p[i]>'9')))
         {
            p[i] = '_';
         }
      }
      p[i] = '\0';
      
      // Grab color
      TMCColorRGBA col1, col2;
      anInstance->GetMainColors(col1, col2);
      
      // Fill data structure for call back
      DAEWriteFacetStruct ws;

      TTreeTransform transform;
      ws.primitive = primitive;
      ws.stream = stream;
      tree->GetGlobalTreeTransform(transform);

      ws.transform = &transform;
      ws.name = p;
      ws.count = count;
      ws.color = col1;
      
      TMCCountedPtr<I3DShObject> anObj;
      anInstance->GetDeformed3DObject(&anObj);
      TMCCountedPtr<I3DShPrimitive> prim;
      anObj->QueryInterface(IID_I3DShPrimitive, (void **) &prim);
      if (prim)
      {
         TMCCountedPtr<FacetMesh> fmesh;
         prim->GetFMesh(0.0, &fmesh);
         FacetMeshFacetIterator facetIterator;
         facetIterator.Initialize(fmesh);
         TFacet3D facet;
         
         uint32 vertexCount = fmesh->fVertices.GetElemCount();
         (*stream) << "\t<library type="GEOMETRY">\n";
         (*stream) << "\t\t<geometry id="" << ws.name << "-lib" name="" << ws.name << "">\n";
         (*stream) << "\t\t\t<mesh>\n";

         //   Write out the position vertices.
         (*stream) << "\t\t\t\t<source id="" << ws.name << "-lib-Position">\n";
         (*stream) << "\t\t\t\t\t<float_array id="" << ws.name << "-lib-Position-array" count="" << vertexCount * 3 << "">\n";

         for (uint32 vertexIndex = 0; vertexIndex < vertexCount; ++vertexIndex)
         {
            (*stream) << "\t\t\t\t\t\t";
            WriteVertex(stream, fmesh->fVertices[vertexIndex]);
            (*stream) << "\n";
         }
         (*stream) << "\t\t\t\t\t</float_array>\n";
         (*stream) << "\t\t\t\t<technique profile="COMMON">\n";
         (*stream) << "\t\t\t\t\t<accessor source="#" << ws.name << "-lib-Position-array" count="" << vertexCount << "" stride="3">\n";
         (*stream) << "\t\t\t\t\t\t<param name="X" type="float" flow="OUT"/>\n";
         (*stream) << "\t\t\t\t\t\t<param name="Y" type="float" flow="OUT"/>\n";
         (*stream) << "\t\t\t\t\t\t<param name="Z" type="float" flow="OUT"/>\n";
         (*stream) << "\t\t\t\t\t</accessor>\n";
         (*stream) << "\t\t\t\t</technique>\n";
         (*stream) << "\t\t\t</source>\n";

         (*stream) << "\t\t\t\t<vertices id="" << ws.name << "-lib-Vertex">\n";
         (*stream) << "\t\t\t\t\t<input semantic="POSITION" source="#" << ws.name << "-lib-Position"/>\n";
         (*stream) << "\t\t\t\t</vertices>\n";

         uint32 facetCount = fmesh->fFacets.GetElemCount();
         (*stream) << "\t\t\t\t<polygons material="#default" count="" << facetCount << "">\n";
         (*stream) << "\t\t\t\t\t<input semantic="VERTEX" idx="0" source="#" << ws.name << "-lib-Vertex"/>\n";

         for (uint32 facetIndex = 0; facetIndex < facetCount; ++facetIndex)
         {
            (*stream) << "\t\t\t\t\t<p>" << fmesh->fFacets[facetIndex].pt1 << " " << fmesh->fFacets[facetIndex].pt2 << " " << fmesh->fFacets[facetIndex].pt3 << "</p>\n";
         }

         (*stream) << "\t\t\t\t</polygons>\n";
         (*stream) << "\t\t\t</mesh>\n";
         (*stream) << "\t\t</geometry>\n";
         (*stream) << "\t</library>\n";
      }
   }
