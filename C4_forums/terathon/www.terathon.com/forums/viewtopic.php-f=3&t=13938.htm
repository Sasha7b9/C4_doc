Voxel Count Calculator

by GameWizard » 23 Oct 2014, 21:04

If the total voxel count for a terrain block is too high, the ability to use the editor to work with the terrain becomes impacted. Rather than continually having to use a calculator to determine what terrain parameters will keep the total voxels small enough, what will work for different resolutions, numbers of extra vertical voxels, etc. I created a web page to do so. I can play with the various parameters using sliders or text input to see how they affect the resolution and total number of voxels. Or, I can fix the resolution to what I need for a block and see what parameter combinations would work.

I think I have it correct. But, post any problems. And, if you make any improvements or add any functionality, please post a new version. I do not know of a file type that can be uploaded, so here is the .html

Code: Select all
<!doctype html>
<html>
<head>
<meta charset="UTF-8">
<title>Voxel Count Calculation</title>
<script>
//===================================================================
//
//  Assuming a square terrain block and square height map image.
//  Assuming terrain resolution = terrain block size / height map image size.
//  Assuming total number of voxels = ((terrain height / terrain resolution) + extra height voxel count + extra depth voxels count) * image size squared.
//
//===================================================================

function UpdateBlockSizeText()  // called when slider changes
{
   document.getElementById("BlockSizeTextId").value = document.getElementById("BlockSizeId").value;
   if(document.getElementById("CheckBoxId").checked)
   {
      var blockSize = parseInt(document.getElementById("BlockSizeTextId").value);
      var fixedResolution   = parseInt(document.getElementById("FixedResolutionTextId").value);
        var newImageSize = blockSize / fixedResolution;
      document.getElementById("ImageSizeTextId").value = newImageSize.toString();
      document.getElementById("ImageSizeId").value = newImageSize;
   }
   UpdateTotalVoxelCount();
}

function UpdateBlockSizeRange()  // called when text block changes
{
   document.getElementById("BlockSizeId").value = document.getElementById("BlockSizeTextId").value;
   if(document.getElementById("CheckBoxId").checked)
   {
      var blockSize = parseInt(document.getElementById("BlockSizeTextId").value);
      var fixedResolution   = parseInt(document.getElementById("FixedResolutionTextId").value);
        var newImageSize = blockSize / fixedResolution;
      document.getElementById("ImageSizeTextId").value = newImageSize.toString();
      document.getElementById("ImageSizeId").value = newImageSize;
   }
   UpdateTotalVoxelCount();
}
//==================================================================
function UpdateImageSizeText()  //called when slider changes
{
   document.getElementById("ImageSizeTextId").value = document.getElementById("ImageSizeId").value;
   if(document.getElementById("CheckBoxId").checked)
   {
      var imageSize = parseInt(document.getElementById("ImageSizeTextId").value);
      var fixedResolution   = parseInt(document.getElementById("FixedResolutionTextId").value);
        var newBlockSize = imageSize * fixedResolution;
      document.getElementById("BlockSizeTextId").value = newBlockSize.toString();
      document.getElementById("BlockSizeTId").value = newBlockSize.toString();
   }   
   UpdateTotalVoxelCount();
}

function UpdateImageSizeRange()  // called when text block changes
{
   //alert("update");
   document.getElementById("ImageSizeId").value = document.getElementById("ImageSizeTextId").value;
   if(document.getElementById("CheckBoxId").checked)
   {
      var imageSize = parseInt(document.getElementById("ImageSizeTextId").value);
      var fixedResolution   = parseInt(document.getElementById("FixedResolutionTextId").value);
        var newBlockSize = imageSize * fixedResolution;
      document.getElementById("BlockSizeTextId").value = newBlockSize.toString();
      document.getElementById("BlockSizeTextId").value = newBlockSize.toString();
   }   
   UpdateTotalVoxelCount();
}

//==================================================================

function UpdateBlockHeightText()
{
   document.getElementById("BlockHeightTextId").value = document.getElementById("BlockHeightId").value;
   UpdateTotalVoxelCount();
}

function UpdateBlockHeightRange()
{
   document.getElementById("BlockHeightId").value = document.getElementById("BlockHeightTextId").value;
   UpdateTotalVoxelCount();
}

//==================================================================

function UpdateExtraHeightVoxelsText()
{
   document.getElementById("ExtraHeightVoxelsTextId").value = document.getElementById("ExtraHeightVoxelsId").value;
   UpdateTotalVoxelCount();
}

function UpdateExtraHeightVoxelsRange()
{
   document.getElementById("ExtraHeightVoxelsId").value = document.getElementById("ExtraHeightVoxelsTextId").value;
   UpdateTotalVoxelCount();
}
//==================================================================

function UpdateExtraDepthVoxelsText()
{
   //alert("update");
   document.getElementById("ExtraDepthVoxelsTextId").value = document.getElementById("ExtraDepthVoxelsId").value;
   UpdateTotalVoxelCount();

}

function UpdateExtraDepthVoxelsRange()
{
   //alert("update");
   document.getElementById("ExtraDepthVoxelsId").value = document.getElementById("ExtraDepthVoxelsTextId").value;
   UpdateTotalVoxelCount();

}

//==================================================================

function CheckboxChanged()
{
   UpdateBlockSizeText();
   UpdateBlockSizeRange();
}
//==================================================================

function FixedResolutionChanged()
{
   UpdateBlockSizeText();
   UpdateBlockSizeRange();
}
//==================================================================
function UpdateTotalVoxelCount()
{
   var blockSize         = parseInt(document.getElementById("BlockSizeTextId").value);
   var imageSize         = parseInt(document.getElementById("ImageSizeTextId").value);
   var blockHeight       = parseInt(document.getElementById("BlockHeightTextId").value);
   var extraHeightVoxels = parseInt(document.getElementById("ExtraHeightVoxelsTextId").value);
   var extraDepthVoxels  = parseInt(document.getElementById("ExtraDepthVoxelsTextId").value);
      
   
    var terrainResolution = blockSize / imageSize;
   document.getElementById("TerrainResolutionTextId").value = terrainResolution.toFixed(2);
    var totalVoxelHeight = (blockHeight / terrainResolution) + extraHeightVoxels + extraDepthVoxels;
   var totalVoxelCount = totalVoxelHeight * imageSize * imageSize;
   
   var totalCountString = totalVoxelCount.toFixed(0);
   var totalCount = parseInt(totalCountString);
   if(totalCount > 999999)
   {
      totalCount = totalCount/1000000;
      totalCountString = totalCount.toFixed(3) + " mb";
   }
   else if(totalCount > 999)
   {
      totalCount = totalCount / 1000;
      totalCountString = totalCount.toFixed(3) + " kb";
   }
   else
   {
      totalCountString = totalCount.toFixed(0) + " bytes";
   }
   
    document.getElementById("TotalCountTextId").value = totalCountString;
}
</script>

</head>
