Re: Message arriving empty

by igor » 21 Oct 2010, 22:38

Its pretty simple and is basically like the other message classes you have in the demo. To debug I disabled the mutex and ran two instances on my machine, taking turns to run the client and host in through visual studio in debug mode (my machine isnt powerful enough to run two instances of visual studio and two instances of C4 to debug both simultaneously).

Multiplayer.h:
Code: Select all
class ClientLoginMessage : public Message
{
   friend class Game;

private:

   String<>   username;
   String<>   password;

   ClientLoginMessage();

public:

   ClientLoginMessage(const char *user, const char *pass);
   ~ClientLoginMessage();

   const char *GetUsername(void) const
   {
      return (username);
   }

   const char *GetPassword(void) const
   {
      return (password);
   }

   void Compress(Compressor& data) const;
   bool Decompress(Decompressor& data);

   bool HandleMessage(Player *sender) const;
};


Multiplayer.cpp:
Code: Select all
ClientLoginMessage::ClientLoginMessage() : Message(kMessageClientLogin)
{
}

ClientLoginMessage::ClientLoginMessage(const char *user, const char *pass) : Message(kMessageClientLogin)
{
   username = user;
   password = pass;
}

ClientLoginMessage::~ClientLoginMessage()
{
}

void ClientLoginMessage::Compress(Compressor& data) const
{
   data << username;
   data << password;
}

bool ClientLoginMessage::Decompress(Decompressor& data)
{
   data >> username;
   data >> password;
   return (true);
}

bool ClientLoginMessage::HandleMessage(Player *sender) const
{
      Actor *participant = nullptr;
      participant = TheGame->GetFirstParticipant();
      bool validUser = false;

      if (participant)
      {
         while (participant && validUser == false)
         {
            if (Text::CompareText(participant ->GetName(),username) &&
               Text::CompareText(participant ->GetPassword(),password))
            {
               validUser = true;
            }

            participant = participant->Next();
         }

         Actor *director = TheGame->GetDirector();
         if (director)
         {
            if (Text::CompareText(director->GetName(),username) &&
               Text::CompareText(director->GetPassword(),password))
            {
               validUser = true;
            }
         }
      }

      ServerAuthenticationMessage authMessage(validUser);
      TheMessageMgr->SendMessage(sender->GetPlayerKey(), authMessage);

   return (true);
}
