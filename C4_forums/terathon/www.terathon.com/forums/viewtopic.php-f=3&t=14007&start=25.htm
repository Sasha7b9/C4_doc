I got it working and its very cool. The flag bounces off walls as well. The CarriedFlagController needs some settings for the RigidBodyController, but I don't know what they should be.


Code: Select all
class CarriedFlagController : public C4::RigidBodyController
{
   public:

      CarriedFlagController() : C4::RigidBodyController(C4::kControllerGeneric)
      {
      }
};


void GetJointTransform(C4::Transform4D *transform)
{
   float modelAltitude;

   ZedC4::GetControllerAltitude(owner, &modelAltitude);

   C4::Point3D position;
   C4::Vector2D u = C4::CosSin(modelAltitude);
   C4::Vector2D t = {1, 0};

   C4::Vector3D view = C4::Vector3D(t.x, t.y, u.y);
   C4::Vector3D right = C4::Vector3D(t.y, -t.x, 0.0F);
   C4::Vector3D down = view % right;
   float d = 0.375F + modelAltitude * 0.1F;
   //position = right * 0.25F + view * d;
   position = right * 2.25F + view * d;
   position.z += 1.15F;

   transform->Set(view, -right, -down, position);
}


void ClassName::Spawn(bool onGround, const C4::Point3D& position)
{
   // Spawn the flag
   C4::Model *model = C4::Model::Get(C4::kModelEricFlag);
   C4::World *world = owner->GetWorld();

   if (onGround)
   {
      model->SetController(new C4::CollectableController());

      C4::Transform4D transform = owner->GetNodeTransform();
      transform.SetTranslation(position);
      model->SetNodeTransform(transform);
      world->AddNewNode(model);
   }
   else
   {
      CarriedFlagController *carriedFlagController = new CarriedFlagController();
      model->SetController(carriedFlagController);

      C4::Transform4D transform;
      GetJointTransform(&transform);

      C4::Point3D offset = transform.GetTranslation();

      transform.SetTranslation(owner->GetWorldTransform().GetTranslation() + offset + C4::Point3D(0.0001F, 0.0F, 0.0F));
      model->SetNodeTransform(transform);
      world->AddNewNode(model);

      delete joint;
      joint = new C4::SphericalJoint();
      transform.SetTranslation(owner->GetWorldTransform().GetTranslation() + offset);
      joint->SetNodeTransform(transform);
      joint->AddConnector(C4::kConnectorKeyBody1, model);
      world->AddNewNode(joint);

      carriedFlagController->Wake();
   }

   return model;
}


void ClassName::Move(void)
{
   if (HasFlag())
   {
      if (joint.GetTarget())
      {
         C4::Transform4D transform;
         GetJointTransform(&transform);

         transform.SetTranslation(owner->GetWorldTransform().GetTranslation() + transform.GetTranslation());
         joint->SetNodeTransform(transform);
         joint->Invalidate();
      }
   }
}
