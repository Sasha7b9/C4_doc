void Game::SpawnPlayers()
{
   if(TheMessageMgr->Server())
   {
      World *world = TheWorldMgr->GetWorld();
      
      if(world)
      {
         GamePlayer *p = static_cast<GamePlayer *>(TheMessageMgr->GetFirstPlayer());
         while(p)
         {
            if(p != TheMessageMgr->GetLocalPlayer())
            {
               long controllerIndex = world->NewControllerIndex();

               Point3D pos(0.0F, 0.0F, 1.0F);
               
               SpawnBox(p, controllerIndex, world, pos);

               String<100> report("Telling player: ");
               report += p->GetPlayerKey();
               report += " to spawn box w/controller index: ";
               report += controllerIndex;
               Engine::Report(report);

               TheMessageMgr->SendMessageClients(BoxSpawnMessage(p->GetPlayerKey(), controllerIndex, pos));
            }   
            
            p = p->Next();
         }

void Game::SpawnBox(GamePlayer *player, long controllerIndex, World *world, Point3D position)
{
   BoxController *controller = new BoxController;
   player->SetPlayerController(controller);
   controller->SetControllerIndex(controllerIndex);

   Model *model = Model::Get(C4::kModelBox);
   model->SetController(controller);

   Zone *zone = world->FindZone(position);
   zone->AddNewSubnode(model);

   String<100> report("Spawning box w/controller index: ");
   report += controllerIndex;
   report += " for player: ";
   report += player->GetPlayerKey();
   Engine::Report(report);

   model->SetNodePosition(Point3D(0.0F, 0.0F, 2.0F));
   model->Invalidate();
   model->Update();
}

void Game::MoveBox()
{
   if(!TheMessageMgr->Server())
   {
      GamePlayer *p = static_cast<GamePlayer *>(TheMessageMgr->GetLocalPlayer());
      String<100> report("Telling server to move box, my controller index is: ");
      report += p->GetPlayerController()->GetControllerIndex();
      Engine::Report(report);

      TheMessageMgr->SendMessage(C4::kPlayerServer, BoxMoveMessage(C4::kMovementUp));
   }
}

      GamePlayer *player = static_cast<GamePlayer *> (sender);
      BoxController *controller = player->GetPlayerController();
      if(controller)
      {
         String<100> report("Player: ");

         controller->SetMovementFlags(movementFlags);

ControllerMessage *BoxController::ConstructMessage(ControllerMessageType type) const
{
   Engine::Report("Controller Message Constructed");

   switch(type)
   {
   case kMessageBoxUpdate:
      {
         Engine::Report("Box Update Message Constructed");
         return (new BoxControllerMessage(this->GetControllerIndex()));
      }
   default:
      {
         break;
      }
   }

   return Controller::ConstructMessage(type);
}

void BoxController::ReceiveMessage(const ControllerMessage *message)
{
   Engine::Report("Controller Message Received");

   switch(message->GetMessageType())
   {
   case kMessageBoxUpdate:
      {
         Engine::Report("Box Update Message Received");
         break;
      }
   default:
      {
         break;
      }
   }
   Controller::ReceiveMessage(message);
}

void BoxController::Move()
{
   Model *model = static_cast<Model *>(this->GetTargetNode());
   if(model)
   {
      if((movementFlags & C4::kMovementUp) > 0)
      {
         Transform4D orig = model->GetNodeTransform();
         float x;
         float y;
         float z;

         orig.GetEulerAngles(&x, &y, &z);

         z += 1.0F * (TheTimeMgr->GetFloatDeltaTime() / 1000.0F);

         orig.SetEulerAngles(x,y,z);

         model->SetNodeTransform(orig);
         model->Invalidate();
         model->Update();

         if(TheMessageMgr->Server())
         {
            BoxControllerMessage message(this->GetControllerIndex(), orig.GetTranslation(), Point3D(x,y,z));
            TheMessageMgr->SendMessageAll(message);
         }
      }
   }
}
