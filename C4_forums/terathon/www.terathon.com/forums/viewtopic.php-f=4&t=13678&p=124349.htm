by cboyce » 20 May 2014, 10:49

I've been working on a port of the NGplant tool in to C4, and had figure out how to create meshes from vertices. Here's the method I have to redraw the mesh. Keep in mind this was my own adaptation of what I found done in the engine, and there could very well be better ways to do this. I can send you all the code if this doesn't point you in the right direction.
Code: Select all
void TreeCreatorPage::redrawrTree(){
   if (treeMesh != nullptr && treeMesh->GetSubnodeCount() > 0){
      treeMesh->PurgeSubtree();
   }
   MaterialObject *material = MaterialObject::Get("NGPlant/Materials/TestGrid01");
   static_cast<WorldViewportWidget*>(FindWidget("viewport"))->GetViewportWorld()->GetRootNode()->AddNewSubnode(treeMesh);


   P3DBranchModel * last = rootModel->GetPlantBase();




   P3DHLIPlantTemplate plantTemplate(rootModel);
   P3DHLIPlantInstance * PlantInstance = plantTemplate.CreateInstance();




   unsigned_int16 groupCount = plantTemplate.GetGroupCount();
   for (unsigned_int16 groupIndex = 0; groupIndex < groupCount; groupIndex++){

      //Number of elements in index buffer required to describe branch geometry
      unsigned_int32 indexCount = plantTemplate.GetIndexCount(groupIndex, P3D_TRIANGLE_LIST);
      TheEngine->Report(String<>("Number of index buffer elements") + Text::IntegerToString(indexCount));

      unsigned_int32 branchCount = PlantInstance->GetBranchCount(groupIndex);
      TheEngine->Report(String<>("Branch count ") + Text::IntegerToString(branchCount));
      Array<unsigned_int32> indexBuffer(indexCount*branchCount);


      List<GeometrySurface> * surfaceList = new List<GeometrySurface>();
      unsigned_int32 attrCount = PlantInstance->GetVAttrCountI(groupIndex);
      Array<float> positionBuffer(attrCount*sizeof(float)* 3);
      Array<float> texCoordBuffer(attrCount*sizeof(float)* 2);

      P3DHLIVAttrBuffers vAttrBuffers;
      vAttrBuffers.AddAttr(P3D_ATTR_VERTEX, positionBuffer, 0, sizeof(float)* 3);
      vAttrBuffers.AddAttr(P3D_ATTR_TEXCOORD0, texCoordBuffer, 0, sizeof(float)* 2);
      PlantInstance->FillVAttrBuffersI(&vAttrBuffers, groupIndex);

      unsigned_int32 branchAttrCount = plantTemplate.GetVAttrCountI(groupIndex);
      
      Array<MaterialObject *> materialArray(branchCount);
      

      material->SetMaterialFlags(material->GetMaterialFlags() | kMaterialTwoSided);
      

      for (unsigned_int16 branchIndex = 0; branchIndex < branchCount; branchIndex++){
         plantTemplate.FillIndexBuffer(
            &indexBuffer[branchIndex*indexCount],
            groupIndex,
            P3D_TRIANGLE_LIST,
            P3D_UNSIGNED_INT
            , branchAttrCount*branchIndex);


         GeometryPolygon *polygon;
         GeometryVertex * vertex;
         float x, y, z, r, s;

         unsigned_int32 triangleStartIndex;
         GeometrySurface * surface = new GeometrySurface;
         for (unsigned_int32 i = 0; i < indexCount; i += 3){
            polygon = new GeometryPolygon;
            for (int16 j = 0; j < 3; j++){
               vertex = new GeometryVertex();
               triangleStartIndex = indexBuffer[i + j + branchIndex*indexCount];
               x = positionBuffer[triangleStartIndex * 3];
               y = positionBuffer[triangleStartIndex * 3 + 1];
               z = positionBuffer[triangleStartIndex * 3 + 2];
               r = texCoordBuffer[triangleStartIndex * 2];
               s = texCoordBuffer[triangleStartIndex * 2 + 1];
               vertex->position.Set(x, y, z);
               vertex->texcoord[0].Set(s, r);
               polygon->vertexList.Append(vertex);

            }
            surface->polygonList.Append(polygon);
         }
         
         
         materialArray.AddElement(material);
         
         surfaceList->Append(surface);
   
      }


      const List<GeometrySurface> *surfaceTable = surfaceList;
      GenericGeometry * treePartMesh = new GenericGeometry(1, &surfaceTable, materialArray);
      


      Transform4D transform;
      transform.SetIdentity();
      transform.SetEulerAngles(C4::K::radians * 90.0f, 0.0f, 0.0f);
      treePartMesh->SetNodeTransform(transform);

      treeMesh->AddNewSubnode(treePartMesh);
      treePartMesh->Preprocess();

   
      delete surfaceList;
      
   
   }
   material->Release();
   delete PlantInstance;
   
}

Re: Making a mesh from scratch

by Eric Lengyel » 20 May 2014, 14:53

The whole GeometryVertex / GeometryPolygon structure is very generic and is used for importing meshes from external sources. The engine does optimize meshes created this way so that duplicate vertices are merged, etc. If you're working inside the engine, you can create your vertex and index arrays more efficiently by using the Mesh class directly. There are a bunch of examples in C4Primitives.cpp for creating meshes for various types of geometries like boxes and cylinders. I'll try to get a self-contained example written on the wiki.

Low-level serialization can be done through the Node::PackTree() and Node::UnpackTree() functions.
