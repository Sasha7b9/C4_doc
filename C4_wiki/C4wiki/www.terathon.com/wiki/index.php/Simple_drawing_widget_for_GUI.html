Simple drawing widget for GUI

From C4 Engine Wiki

Jump to: navigation, search

In this example describes a simple widget for software rendering own interface (C4 engine ver.4.0) 

 /*************/
 /* Artist.h  */
 /*************/
 
 #pragma once
 
 class Artist : public C4::Widget {
 
 public:
    Artist(int width, int height);
    void Clear();
    void SetColorBackground(const C4::ColorRGBA &color);
    void SetColorBrush(const C4::ColorRGBA &color);
    void DrawPoint(int x, int y);
 
 private:
    C4::Texture *texture = nullptr;
    unsigned_int32 *imageData = nullptr;
    C4::ColorRGBA colorBackground;
    C4::ColorRGBA colorBrush;
    int width = 0;
    int height = 0;
 };
 /****************/
 /* Artist.cpp   */
 /****************/
 
 #include "Artist.h"
 #include "../Math.h"
 
 using C4::ImageWidget;
 using C4::Vector2D;
 using C4::Widget;
 using C4::Texture;
 using C4::TextureHeader;
 using C4::Rect;
 
 Artist::Artist(int width, int height) : Widget() {
 
    this->width = width;
    this->height = height;
    colorBackground = C4::K::black;
    colorBrush = C4::K::white;
 
    TextureHeader *textureHeader = new TextureHeader();
    textureHeader->textureType = C4::kTexture2D;
    textureHeader->textureFlags = 0;
    textureHeader->colorSemantic = C4::kTextureSemanticDiffuse;
    textureHeader->alphaSemantic = C4::kTextureSemanticTransparency;
    textureHeader->imageFormat = C4::kTextureRGBA8;
    textureHeader->imageWidth = width;
    textureHeader->imageHeight = height;
    textureHeader->imageDepth = 1;
    textureHeader->wrapMode[0] = C4::kTextureClamp;
    textureHeader->wrapMode[1] = C4::kTextureClamp;
    textureHeader->wrapMode[2] = C4::kTextureClamp;
    textureHeader->mipmapCount = 1;
    textureHeader->mipmapDataOffset = 0;
    textureHeader->auxiliaryDataSize = 0;
    textureHeader->auxiliaryDataOffset = 0;
 
    imageData = new unsigned_int32[width * height];
    texture = C4::Texture::Get(textureHeader, imageData);
 
    ImageWidget *image = new ImageWidget(Vector2D(width, height), texture);
 
    Clear();
 
    AppendNewSubnode(image);
 }
 
 void Artist::SetColorBackground(const C4::ColorRGBA &color) {
    colorBackground = color;
 }
 
 void Artist::SetColorBrush(const C4::ColorRGBA &color) {
    colorBrush = color;
 }
 
 void Artist::DrawPoint(int x, int y) {
    imageData[x + y * width] = Math::ColorToUINT(colorBrush);
    texture->UpdateRect(Rect(0, 0, width, height), 0, imageData);
 }
 
 void Artist::Clear() {
    unsigned_int32 colorUINT = Math::ColorToUINT(colorBackground);
    for(int i = 0; i < width * height; i++) {
        imageData[i] = colorUINT;
    }
    texture->UpdateRect(Rect(0, 0, width, height), 0, imageData);
 }

Helper class: 

 #pragma once
 
 class Math {
 
 public:
    static unsigned_int32 ColorToUINT(const C4::ColorRGBA &color);
 };
 unsigned_int32 Math::ColorToUINT(const C4::ColorRGBA &color) {
    unsigned_int32 red = color.red * 255;
    unsigned_int32 green = color.green * 255;
    unsigned_int32 blue = color.blue * 255;
    unsigned_int32 alpha = color.alpha * 255;
    return red | (green << 8) | (blue << 16) | (alpha << 24);
 }

Example of use: 

 Artist *artist = new Artist(300, 300);
 TheInterfaceMgr->AddWidget(artist);
 artist->SetColorBackground(ColorRGBA::blue);
 artist->Clear();
 artist->SetColorBrush(ColorRGBA::green);
 artist->DrawPoint(10, 100);
 artist->SetWidgetPosition(Point3D(300, 300, 0));
