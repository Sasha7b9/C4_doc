Draw Lines As Debugging Help

This article describes how to draw lines for debugging purposes. Beware that this is a rather slow technique and will slow down your application significantly if you use a lot of them so it should not be used in an optimzied program version. The basic code was written by A2K as can seen in this forum post . However this does not work with V3 of the C4 engine anymore, so I altered the code and post it here.

This code must be called from your World::Render() overloaded function. 

void World_Debug_Render::Draw(C4::RenderType t_type, const C4::Array<C4::Point3D>& rcac_points, const C4::Array<C4::ColorRGBA>& c_colours, C4::ColorRGBA c_alpha_colour)
{
    if (rcac_points.GetElementCount() == 0)
        return;
    unsigned_int32 u32_state_flags = kRenderLineSmooth | kRenderDepthTest | kRenderDepthInhibit;
    Renderable c_renderable_lines(t_type, u32_state_flags);
    // Fill the vertexbuffer with all vertices
    //
    VertexBuffer	c_line_vertex_buffer(kVertexBufferAttribute | kVertexBufferDynamic);
    if (!c_line_vertex_buffer.Active())
        c_line_vertex_buffer.Establish(sizeof(Point3D) * rcac_points.GetElementCount());
 
    volatile Point3D *restrict pc_vertex = c_line_vertex_buffer.BeginUpdate<Point3D>();
    for( S32 s32_index = 0; s32_index < rcac_points.GetElementCount() ; s32_index++ )
        pc_vertex[s32_index]		= rcac_points[s32_index];
    c_line_vertex_buffer.EndUpdate();
 
    c_renderable_lines.SetVertexCount(rcac_points.GetElementCount());
    c_renderable_lines.SetVertexBuffer(kVertexBufferAttributeArray, &c_line_vertex_buffer, sizeof(Point3D));
    c_renderable_lines.SetVertexAttributeArray(kArrayPosition, 0, 3);
 
    // Set the line colour
    //
    List<Attribute>		c_attribute_list;
    DiffuseAttribute	c_diffuse_colour(c_colours[0]);
 
    if (c_alpha_colour != K::white)
    {
        // Couldn't get the ambient pass working for alpha blend with c_renderable_lines
        c_renderable_lines.SetShaderFlags(c_renderable_lines.GetShaderFlags() | kShaderAmbientEffect);
        c_renderable_lines.SetAmbientBlendState(BlendState(kBlendSourceAlpha, kBlendOne));
    }
 
    c_attribute_list.Append(& c_diffuse_colour);
    c_renderable_lines.SetMaterialAttributeList( & c_attribute_list);
    c_renderable_lines.SetRenderableFlags(kRenderableFogInhibit);
 
    // Create a list of one renderable and draw it
    //
    List<Renderable> c_render_list;
    c_render_list.Append(&c_renderable_lines);
    TheGraphicsMgr->DrawRenderList(&c_render_list);
    c_render_list.RemoveAll();
    c_attribute_list.RemoveAll();
}


Here is a more elaborate implementation that implements the Box2D b2Draw interface. This method is *much* faster, and can be expected to attain playable framerates with several hundred debug draw calls. This version should be threadsafe.

Header: 

    struct B2DebugVertex {
    	C4::Point3D position;
    	C4::ColorRGBA color;
    };
 
    enum B2DebugOpType {
    	kOpLine,
    	kOpLineStrip,
    	kOpLoop
    };
 
    class B2DebugOp : public C4::ListElement<B2DebugOp> {
    public:
    	B2DebugOpType type;
    	int nVerts;
    	B2DebugVertex verts[1024];
    };
 
    /**
     * B2 Debug drawer
     */
    class B2Debugger : public b2Draw
    {
    	C4::Mutex lock;
    	C4::List<C4::Attribute> attributeList;
    	C4::DiffuseAttribute diffuse;
 
    	C4::List<C4::Renderable> renderList;
 
    	C4::Array<C4::Renderable *, 1024> loopCloseRenders;
    	int curLoopCloseRender;
    	C4::Array<C4::Renderable *, 1024> loopOpenRenders;
    	int curLoopOpenRender;
    	C4::Array<C4::Renderable *, 1024> lineRenders;
    	int curLineRender;
 
    	C4::Array<B2DebugOp *, 1024> debugOps;
    	int curOp;
 
    	C4::List<B2DebugOp> frameOps;
 
    	C4::Renderable *NewRender(int& index, C4::Array<C4::Renderable *, 1024>& array, C4::RenderType type);
    	C4::VertexBuffer *Buffer(C4::Renderable *r);
    	void AppendRender(C4::Renderable *r);
 
    	C4::Renderable *LoadRenderable(B2DebugOp *op);
 
    	B2DebugOp *NewOp();
    	void AppendOp(B2DebugOp *op);
 
    public:
 
    	C4API B2Debugger();
    	C4API ~B2Debugger();
 
    	C4API void Clear();
 
    	C4API void DrawPolygon(const b2Vec2* vertices, int32 vertexCount, const b2Color& color);
 
    	C4API void DrawPolyline(const b2Vec2* vertices, int32 vertexCount, const b2Color& color);
 
    	C4API void DrawSolidPolygon(const b2Vec2* vertices, int32 vertexCount, const b2Color& color);
 
    	C4API void DrawCircle(const b2Vec2& center, float32 radius, const b2Color& color);
 
    	C4API void DrawSolidCircle(const b2Vec2& center, float32 radius, const b2Vec2& axis, const b2Color& color);

Source: 

 
    static const int MAX_B2_DEBUG_VERTS = 1024;
 
    B2Debugger::B2Debugger() : b2Draw(),
     lock(),
     diffuse(C4::K::white)
    {
    	curLineRender = curLoopCloseRender = curLoopOpenRender = 0;
 
    	attributeList.Append(&diffuse);
    }
 
    B2Debugger::~B2Debugger()
    {
    	for (int i = 0; i < loopCloseRenders.GetElementCount(); i++){
    		delete loopCloseRenders[i];
    	}
 
    	for (int i = 0; i < loopOpenRenders.GetElementCount(); i++){
    		delete loopOpenRenders[i];
    	}
 
    	for (int i = 0; i < lineRenders.GetElementCount(); i++){
    		delete lineRenders[i];
    	}
    }
 
    C4::Renderable *B2Debugger::NewRender(int& index, C4::Array<C4::Renderable *, 1024 > & array, C4::RenderType type)
    {
    	lock.Acquire();
    	C4::Renderable *retval;
    	if (array.GetElementCount() <= index) {
    		retval = new C4::Renderable(type, C4::kRenderLineSmooth | C4::kRenderDepthInhibit);
    		retval->SetMaterialAttributeList(&attributeList);
    		retval->SetVertexAttributeArray(C4::kArrayPosition, 0, 3);
    		retval->SetVertexAttributeArray(C4::kArrayColor, sizeof(C4::Point3D), 4);
 
    		C4::VertexBuffer *vBuf = new C4::VertexBuffer(C4::kVertexBufferAttribute | C4::kVertexBufferDynamic);
    		vBuf->Establish(sizeof(B2DebugVertex) * MAX_B2_DEBUG_VERTS);
 
    		retval->SetVertexBuffer(C4::kVertexBufferAttributeArray, vBuf, sizeof(B2DebugVertex));
 
    		array.AddElement(retval);
    	}
    	else {
    		retval = array[index];
    	}
    	index++;
    	lock.Release();
    	return retval;
    }
 
    C4::VertexBuffer *B2Debugger::Buffer(C4::Renderable* r)
    {
    	return const_cast<C4::VertexBuffer*>(r->GetVertexBuffer(C4::kVertexBufferAttributeArray));
    }
 
    void B2Debugger::AppendRender(C4::Renderable* r)
    {
    	lock.Acquire();
    	renderList.Append(r);
    	lock.Release();
    }
 
    B2DebugOp *B2Debugger::NewOp()
    {
    	lock.Acquire();
    	B2DebugOp *retval;
    	if (debugOps.GetElementCount() <= curOp){
    		retval = new B2DebugOp;
    		debugOps.AddElement(retval);
    	}
    	else {
    		retval = debugOps[curOp];
    	}
    	curOp++;
 
    	lock.Release();
    	return retval;
    }
 
    void B2Debugger::AppendOp(B2DebugOp* op)
    {
    	lock.Acquire();
    	frameOps.Append(op);
    	lock.Release();
    }
 
    C4::Renderable *B2Debugger::LoadRenderable(B2DebugOp* op)
    {
    	C4::Renderable *retval;
    	switch (op->type){
    	case kOpLine:
    		retval = NewRender(curLineRender, lineRenders, C4::kRenderLines);
    		break;
    	case kOpLineStrip:
    		retval = NewRender(curLoopOpenRender, loopOpenRenders, C4::kRenderLineStrip);
    		break;
    	case kOpLoop:
    		retval = NewRender(curLoopCloseRender, loopCloseRenders, C4::kRenderLineLoop);
    		break;
    	}
 
    	C4::VertexBuffer *vBuf = Buffer(retval);
    	vBuf->UpdateBuffer(0, sizeof(B2DebugVertex) * op->nVerts, op->verts);
    	retval->SetVertexCount(op->nVerts);
 
    	return retval;
    }
 
    void B2Debugger::DrawPolygon(const b2Vec2* vertices, int32 vertexCount, const b2Color& color)
    {
    	if (GetFlags() == 0) return;
 
    	B2DebugOp *op = NewOp();
    	op->type = kOpLoop;
    	op->nVerts = vertexCount;
 
    	volatile B2DebugVertex *restrict vert = &op->verts[0];
 
    	for (int i = 0; i < vertexCount; i++){
    		vert->position = convB2Point(vertices[i]);
    		vert->color.Set(color.r, color.g, color.b, 1.0f);
    		vert++;
    	}
 
    	AppendOp(op);
    }
 
    void B2Debugger::DrawPolyline(const b2Vec2* vertices, int32 vertexCount, const b2Color& color)
    {
    	if (GetFlags() == 0) return;
 
    	B2DebugOp *op = NewOp();
    	op->type = kOpLineStrip;
    	op->nVerts = vertexCount;
 
    	volatile B2DebugVertex *restrict vert = &op->verts[0];
 
    	for (int i = 0; i < vertexCount; i++){
    		vert->position = convB2Point(vertices[i]);
    		vert->color.Set(color.r, color.g, color.b, 1.0f);
    		vert++;
    	}
 
    	AppendOp(op);
    }
 
    void B2Debugger::DrawSolidPolygon(const b2Vec2* vertices, int32 vertexCount, const b2Color& color)
    {
    	DrawPolygon(vertices, vertexCount, color);
    }
 
    void B2Debugger::DrawCircle(const b2Vec2& center, float32 radius, const b2Color& color)
    {
    	if (GetFlags() == 0) return;
 
    	const int nVerts = 15;
 
    	B2DebugOp *op = NewOp();
    	op->type = kOpLoop;
    	op->nVerts = nVerts;
 
    	volatile B2DebugVertex *restrict vert = &op->verts[0];
 
    	float sliceAngle = (2 * C4::K::pi) / nVerts;
    	for (int i = 0; i < nVerts; i++){
    		float x, y;
    		x = (radius * C4::Cos(sliceAngle * i)) + center.x;
    		y = (radius * C4::Sin(sliceAngle * i)) + center.y;
 
    		vert->position.Set(x, 0, y);
    		vert->color.Set(color.r, color.g, color.b, 1.0f);
    		vert++;
    	}
 
    	AppendOp(op);
    }
 
    void B2Debugger::DrawSolidCircle(const b2Vec2& center, float32 radius, const b2Vec2& axis, const b2Color& color)
    {
    	DrawCircle(center, radius, color);
    	DrawSegment(center, center + (radius * axis), color);
    }
 
    void B2Debugger::DrawSegment(const b2Vec2& p1, const b2Vec2& p2, const b2Color& color)
    {
    	if (GetFlags() == 0) return;
 
    	B2DebugOp *op = NewOp();
    	op->type = kOpLoop;
    	op->nVerts = 2;
 
    	volatile B2DebugVertex *restrict vert = &op->verts[0];
 
    	vert->position = convB2Point(p1);
    	vert->color.Set(color.r, color.g, color.b, 1.0f);
 
    	vert++;
 
    	vert->position = convB2Point(p2);
    	vert->color.Set(color.r, color.g, color.b, 1.0f);
 
    	AppendOp(op);
    }
 
    void B2Debugger::DrawVector(const b2Vec2& p1, const b2Vec2& vec, const b2Color& color)
    {
    	if (GetFlags() == 0) return;
 
    	DrawCircle(p1, 0.15f, color);
    	DrawSegment(p1, p1 + vec, color);
    }
 
    void B2Debugger::DrawTransform(const b2Transform& xf)
    {
    	DrawSegment(xf.p, xf.p + (0.5f * xf.q.GetXAxis()), b2Color(0.0f, 0.802f, 1.0f));
    	DrawSegment(xf.p, xf.p + (0.5f * xf.q.GetYAxis()), b2Color(0.0f, 0.5524f, 1.0f));
    }
 
    void B2Debugger::DrawDebugFrame()
    {
    	tmZoneFiltered(tmCtx, 500, TMZF_NONE, "B2Debugger::Draw");
 
    	B2DebugOp *op = frameOps.First();
    	while (op){
    		C4::Renderable *r = LoadRenderable(op);
    		renderList.Append(r);
    		op = op->Next();
    	}
 
    	frameOps.RemoveAll();
 
    	if (!renderList.Empty()) {
    		glLineWidth(4.0f);
    		C4::TheGraphicsMgr->DrawRenderList(&renderList);
    		glLineWidth(1.0f);
    	}
 
    }
 
    void B2Debugger::Clear()
    { 
    	frameOps.RemoveAll();
    	renderList.RemoveAll();
    	curOp = curLoopCloseRender = curLoopOpenRender = curLineRender = 0;
    }
 
 
    void B2Debugger::SetFlags(uint32 flags) 
    {
    	b2Draw::SetFlags(flags);
    }
 
    uint32 B2Debugger::GetFlags() const
    {
    	return b2Draw::GetFlags();
    }
